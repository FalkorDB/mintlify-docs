Provisions a new Enterprise BYOA Standalone FalkorDB instance in the specified region and cloud provider.

## Authorization

<ParamField header="Authorization" type="string" required>
  Bearer JWT obtained from [Sign in](/api-reference/authentication/signin).
</ParamField>

## Query parameters

<ParamField query="subscriptionId" type="string">
  ID of the subscription. Optional.
</ParamField>

## Body

<ParamField body="cloud_provider" type="string" required>
  Cloud provider to provision the resource on.
</ParamField>

<ParamField body="region" type="string" required>
  Region to provision the resource in.
</ParamField>

<ParamField body="requestParams" type="object" required>
  Instance configuration.

  <Expandable title="requestParams properties">
    <ParamField body="cloud_provider_account_config_id" type="string" required>
      ID of the cloud provider account configuration to use for BYOA.
    </ParamField>

    <ParamField body="cloud_provider_native_network_id" type="string">
      ID of an existing native network in the BYOA account to attach to.
    </ParamField>

    <ParamField body="name" default="My favorite database" type="string">
      Human-readable name for the instance.
    </ParamField>

    <ParamField body="description" default="Description" type="string">
      Description of the instance.
    </ParamField>

    <ParamField body="nodeInstanceType" default="e2-custom-4-8192" type="string" required>
      Compute instance type for the node.
    </ParamField>

    <ParamField body="maxMemory" default="6GB" type="string">
      Maximum memory for the database. Do not exceed the memory of the selected instance type.
    </ParamField>

    <ParamField body="falkordbUser" default="falkordb" type="string" required>
      Default username for the database.
    </ParamField>

    <ParamField body="falkordbPassword" type="string" required>
      Default password for the database. Changing this after deployment restarts the instance.
    </ParamField>

    <ParamField body="enableTLS" default={false} type="boolean" required>
      Enable TLS for the database.
    </ParamField>

    <ParamField body="enableEnterpriseModule" default={false} type="boolean" required>
      Enable the enterprise module.
    </ParamField>

    <ParamField body="AOFPersistenceConfig" default="everysec" type="string">
      AOF persistence configuration.
    </ParamField>

    <ParamField body="RDBPersistenceConfig" default="low" type="string">
      How often to save the RDB file to disk.
    </ParamField>

    <ParamField body="falkorDBCacheSize" default={25} type="number">
      Cache size for the database.
    </ParamField>

    <ParamField body="falkorDBMaxQueuedQueries" default={50} type="number">
      Maximum queued queries.
    </ParamField>

    <ParamField body="falkorDBNodeCreationBuffer" default={16384} type="number">
      Node creation buffer size.
    </ParamField>

    <ParamField body="falkorDBQueryMemCapacity" default={21474836480} type="number">
      Query memory capacity in bytes.
    </ParamField>

    <ParamField body="falkorDBResultSetSize" default={10000} type="number">
      Maximum result set size.
    </ParamField>

    <ParamField body="falkorDBTimeoutDefault" default={0} type="number">
      Default query timeout in ms.
    </ParamField>

    <ParamField body="falkorDBTimeoutMax" default={0} type="number">
      Maximum query timeout in ms.
    </ParamField>
  </Expandable>
</ParamField>

## Response

<ResponseField name="id" type="string">
  ID of the created instance.
</ResponseField>

{
  "id": "instance-abc123"
